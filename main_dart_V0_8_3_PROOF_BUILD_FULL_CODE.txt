String soundTypeToJson(SoundType type) {
  switch (type) {
    case SoundType.speech:
      return 'speech';
    case SoundType.possibleSnore:
      return 'possible_snore';
    case SoundType.unknown:
      return 'unknown';
    case SoundType.impact:
      return 'impact';
    case SoundType.background:
      return 'background';
  }
}

SoundType soundTypeFromJson(dynamic value) {
  final v = value?.toString() ?? '';
  if (v == 'speech') return SoundType.speech;
  if (v == 'possible_snore' || v == 'snore') return SoundType.possibleSnore;
  if (v == 'unknown') return SoundType.unknown;
  if (v == 'impact') return SoundType.impact;
  return SoundType.background;
}

class JsonTools {
  static int i(dynamic value) {
    if (value is int) return value;
    if (value is num) return value.toInt();
    return int.tryParse(value?.toString() ?? '') ?? 0;
  }

  static double d(dynamic value) {
    if (value is double) return value;
    if (value is num) return value.toDouble();
    return double.tryParse(value?.toString() ?? '') ?? 0.0;
  }
}

class SleepGraphPoint {
  final int offsetSeconds;
  final double avgDbfs;
  final double peakDbfs;
  final SoundType type;

  const SleepGraphPoint({
    required this.offsetSeconds,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.type,
  });

  Map<String, dynamic> toJson() => {
        'offsetSeconds': offsetSeconds,
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'type': soundTypeToJson(type),
      };

  factory SleepGraphPoint.fromJson(Map<String, dynamic> json) {
    return SleepGraphPoint(
      offsetSeconds: JsonTools.i(json['offsetSeconds']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      type: soundTypeFromJson(json['type']),
    );
  }
}

class SleepEventSegment {
  final int id;
  final int startSeconds;
  final int endSeconds;
  final SoundType type;
  final double avgDbfs;
  final double peakDbfs;
  final String detail;

  const SleepEventSegment({
    required this.id,
    required this.startSeconds,
    required this.endSeconds,
    required this.type,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.detail,
  });

  int get durationSeconds => math.max(1, endSeconds - startSeconds);

  String get rangeText => '${formatDuration(startSeconds)} - ${formatDuration(endSeconds)}';

  Map<String, dynamic> toJson() => {
        'id': id,
        'startSeconds': startSeconds,
        'endSeconds': endSeconds,
        'type': soundTypeToJson(type),
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'detail': detail,
      };

  factory SleepEventSegment.fromJson(Map<String, dynamic> json) {
    return SleepEventSegment(
      id: JsonTools.i(json['id']),
      startSeconds: JsonTools.i(json['startSeconds']),
      endSeconds: JsonTools.i(json['endSeconds']),
      type: soundTypeFromJson(json['type']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      detail: json['detail']?.toString() ?? '',
    );
  }

  SleepEventSegment copyWith({
    int? id,
    int? startSeconds,
    int? endSeconds,
    SoundType? type,
    double? avgDbfs,
    double? peakDbfs,
    String? detail,
  }) {
    return SleepEventSegment(
      id: id ?? this.id,
      startSeconds: startSeconds ?? this.startSeconds,
      endSeconds: endSeconds ?? this.endSeconds,
      type: type ?? this.type,
      avgDbfs: avgDbfs ?? this.avgDbfs,
      peakDbfs: peakDbfs ?? this.peakDbfs,
      detail: detail ?? this.detail,
    );
  }
}

class SleepRecordingSession {
  final DateTime startedAt;
  final DateTime endedAt;
  final int durationSeconds;
  final String audioPath;
  final int fileSizeBytes;
  final int speech;
  final int possibleSnore;
  final int impact;
  final int unknown;
  final double avgDbfs;
  final double peakDbfs;
  final String analysisNote;
  final List<SleepGraphPoint> graphPoints;
  final List<SleepEventSegment> segments;

  const SleepRecordingSession({
    required this.startedAt,
    required this.endedAt,
    required this.durationSeconds,
    required this.audioPath,
    required this.fileSizeBytes,
    required this.speech,
    required this.possibleSnore,
    required this.impact,
    required this.unknown,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.analysisNote,
    required this.graphPoints,
    required this.segments,
  });

  int get totalEvents => segments.length;

  String get durationText => formatDuration(durationSeconds);

  String get avgDbText {
    if (avgDbfs <= -120) return '-- dBFS';
    return '${avgDbfs.toStringAsFixed(1)} dBFS';
  }

  String get peakDbText {
    if (peakDbfs <= -120) return '-- dBFS';
    return '${peakDbfs.toStringAsFixed(1)} dBFS';
  }

  String get fileSizeText {
    if (fileSizeBytes <= 0) return '--';
    final mb = fileSizeBytes / (1024 * 1024);
    if (mb >= 1) return '${mb.toStringAsFixed(2)} MB';
    return '${(fileSizeBytes / 1024).toStringAsFixed(1)} KB';
  }

  Map<String, dynamic> toJson() => {
        'startedAt': startedAt.toIso8601String(),
        'endedAt': endedAt.toIso8601String(),
        'durationSeconds': durationSeconds,
        'audioPath': audioPath,
        'fileSizeBytes': fileSizeBytes,
        'speech': speech,
        'possibleSnore': possibleSnore,
        'impact': impact,
        'unknown': unknown,
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'analysisNote': analysisNote,
        'graphPoints': graphPoints.map((e) => e.toJson()).toList(),
        'segments': segments.map((e) => e.toJson()).toList(),
      };

  factory SleepRecordingSession.fromJson(Map<String, dynamic> json) {
    final gp = json['graphPoints'];
    final seg = json['segments'];

    return SleepRecordingSession(
      startedAt: DateTime.tryParse(json['startedAt']?.toString() ?? '') ??
          DateTime.fromMillisecondsSinceEpoch(0),
      endedAt: DateTime.tryParse(json['endedAt']?.toString() ?? '') ??
          DateTime.fromMillisecondsSinceEpoch(0),
      durationSeconds: JsonTools.i(json['durationSeconds']),
      audioPath: json['audioPath']?.toString() ?? '',
      fileSizeBytes: JsonTools.i(json['fileSizeBytes']),
      speech: JsonTools.i(json['speech']),
      possibleSnore: JsonTools.i(json['possibleSnore']),
      impact: JsonTools.i(json['impact']),
      unknown: JsonTools.i(json['unknown']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      analysisNote: json['analysisNote']?.toString() ?? '',
      graphPoints: gp is List
          ? gp
              .whereType<Map>()
              .map(
                (e) => SleepGraphPoint.fromJson(
                  e.map((k, v) => MapEntry(k.toString(), v)),
                ),
              )
              .toList()
          : const [],
      segments: seg is List
          ? seg
              .whereType<Map>()
              .map(
                (e) => SleepEventSegment.fromJson(
                  e.map((k, v) => MapEntry(k.toString(), v)),
                ),
              )
              .toList()
          : const [],
    );
  }
}

class SleepSessionStore {
  static const _fileName = 'lifecheck_sleep_sessions_v083_proof.json';

  static Future<File> _file() async {
    final directory = await getApplicationDocumentsDirectory();
    return File('${directory.path}/$_fileName');
  }

  static Future<List<SleepRecordingSession>> loadAll() async {
    try {
      final file = await _file();
      if (!await file.exists()) return [];
      final raw = await file.readAsString();
      if (raw.trim().isEmpty) return [];

      final decoded = jsonDecode(raw);
      if (decoded is! List) return [];

      return decoded
          .whereType<Map>()
          .map(
            (e) => SleepRecordingSession.fromJson(
              e.map((k, v) => MapEntry(k.toString(), v)),
            ),
          )
          .toList();
    } catch (_) {
      return [];
    }
  }

  static Future<SleepRecordingSession?> loadLatest() async {
    final sessions = await loadAll();
    if (sessions.isEmpty) return null;
    return sessions.last;
  }

  static Future<void> append(SleepRecordingSession session) async {
    final file = await _file();
    final sessions = await loadAll();
    sessions.add(session);

    // Keep last 30 sessions only for test build.
    final trimmed = sessions.length > 30
        ? sessions.sublist(sessions.length - 30)
        : sessions;

    await file.writeAsString(
      const JsonEncoder.withIndent('  ').convert(
        trimmed.map((e) => e.toJson()).toList(),
      ),
      flush: true,
    );
  }
}

class LifeCheckSleepApp extends StatefulWidget {
  const LifeCheckSleepApp({super.key});

  @override
  State<LifeCheckSleepApp> createState() => _LifeCheckSleepAppState();
}

class _LifeCheckSleepAppState extends State<LifeCheckSleepApp> {
  int _index = 0;
  int _recordStartToken = 0;
  SleepRecordingSession? _latestSession;

  @override
  void initState() {
    super.initState();
    _loadLatest();
  }

  Future<void> _loadLatest() async {
    final session = await SleepSessionStore.loadLatest();
    if (!mounted) return;
    setState(() => _latestSession = session);
  }

  void _startRecordingFlow() {
    setState(() {
      _recordStartToken++;
      _index = 1;
    });
  }

  void _handleSessionSaved(SleepRecordingSession session) {
    setState(() {
      _latestSession = session;
      _index = 2;
    });
  }

  @override
  Widget build(BuildContext context) {
    final screens = [
      HomeScreen(onStart: _startRecordingFlow),
      RecordingScreen(
        startToken: _recordStartToken,
        onSessionSaved: _handleSessionSaved,
        onViewReport: () => setState(() => _index = 2),
      ),
      ReportScreen(
        session: _latestSession,
        onRecordAgain: _startRecordingFlow,
      ),
      SettingsScreen(onRecordAgain: _startRecordingFlow),
    ];

    return MaterialApp(
      title: 'LifeCheck Sleep',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        useMaterial3: true,
        brightness: Brightness.dark,
        scaffoldBackgroundColor: AppColors.navy,
        fontFamily: 'Roboto',
      ),
      home: Scaffold(
        backgroundColor: AppColors.navy,
        body: IndexedStack(index: _index, children: screens),
        bottomNavigationBar: NavigationBarTheme(
          data: NavigationBarThemeData(
            backgroundColor: AppColors.navy2,
            indicatorColor: AppColors.blue.withOpacity(0.18),
            labelTextStyle: WidgetStateProperty.resolveWith(
              (states) => TextStyle(
                color: states.contains(WidgetState.selected)

          child: NavigationBar(

class SleepingHeroPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final moonPaint = Paint()..color = AppColors.blue.withOpacity(0.22);
    canvas.drawCircle(Offset(size.width - 50, 48), 38, moonPaint);
    canvas.drawCircle(
      Offset(size.width - 35, 37),
      34,
      Paint()..color = const Color(0xFF071426),
    );

    final starPaint = Paint()..color = AppColors.white.withOpacity(0.45);
    final stars = [
      Offset(size.width * 0.20, 34),
      Offset(size.width * 0.43, 26),
      Offset(size.width * 0.67, 78),
      Offset(size.width * 0.86, 120),
      Offset(size.width * 0.13, 106),
    ];
    for (final p in stars) {
      canvas.drawCircle(p, 1.35, starPaint);
    }

    final bedY = size.height * 0.64;
    final bedPaint = Paint()..color = AppColors.white.withOpacity(0.10);
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(18, bedY, size.width - 36, 54),
        const Radius.circular(27),
      ),
      bedPaint,
    );

    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(38, bedY - 32, 102, 38),
        const Radius.circular(18),
      ),
      Paint()..color = AppColors.white.withOpacity(0.86),
    );

    canvas.drawCircle(
      Offset(110, bedY - 12),
      19,
      Paint()..color = const Color(0xFFE8B99E),
    );

    final hair = Paint()
      ..color = const Color(0xFF1C1720)
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeWidth = 10;
    canvas.drawArc(
      Rect.fromCircle(center: Offset(103, bedY - 16), radius: 22),
      math.pi * 0.74,
      math.pi * 1.38,
      false,
      hair,
    );

    final blanketRect = Rect.fromLTWH(70, bedY - 4, size.width * 0.64, 66);
    canvas.drawRRect(
      RRect.fromRectAndRadius(blanketRect, const Radius.circular(32)),
      Paint()
        ..shader = const LinearGradient(
          colors: [Color(0xFF2C6DF2), Color(0xFF7C5CFF)],
        ).createShader(blanketRect),
    );

    // phone near bed
    final phoneRect = Rect.fromLTWH(size.width - 86, bedY + 16, 44, 24);
    canvas.drawRRect(
      RRect.fromRectAndRadius(phoneRect, const Radius.circular(7)),
      Paint()..color = AppColors.navy3.withOpacity(0.88),
    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(phoneRect.deflate(3), const Radius.circular(5)),
      Paint()..color = AppColors.blue.withOpacity(0.28),
    );

    // subtle waveform above bed
    final wavePaint = Paint()
      ..color = AppColors.cyan.withOpacity(0.55)
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;
    final baseY = bedY - 55;
    for (int i = 0; i < 34; i++) {
      final x = size.width * 0.42 + i * 5.2;
      final amp = 4 + math.sin(i * 0.65).abs() * 13;
      canvas.drawLine(Offset(x, baseY - amp), Offset(x, baseY + amp), wavePaint);
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class RecordingScreen extends StatefulWidget {
  final int startToken;
  final ValueChanged<SleepRecordingSession> onSessionSaved;
  final VoidCallback onViewReport;

  const RecordingScreen({
    super.key,
    required this.startToken,
    required this.onSessionSaved,
    required this.onViewReport,
  });

  @override
  State<RecordingScreen> createState() => _RecordingScreenState();
}

class _RecordingScreenState extends State<RecordingScreen> {
  final AudioRecorder _recorder = AudioRecorder();

  Timer? _timer;
  Timer? _meterTimer;
  StreamSubscription<Amplitude>? _amplitudeSub;

  String? _recordingPath;
  int _seconds = 0;
  bool _isRecording = false;
  bool _isAnalyzing = false;
  bool _savedCurrent = false;
  String? _error;
  int _lastToken = 0;
  DateTime? _startedAt;
  int _fileSizeBytes = 0;

  double _avgLiveDb = 0;
  double _peakLiveDb = -160;
  double _dbSum = 0;
  int _dbSamples = 0;
  DateTime? _lastAmplitudeAt;

  @override
  void didUpdateWidget(covariant RecordingScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.startToken != 0 && widget.startToken != _lastToken) {
      _lastToken = widget.startToken;
      _startRecording();
    }
  }

  Future<void> _startRecording() async {
    _timer?.cancel();
    _meterTimer?.cancel();
    await _amplitudeSub?.cancel();

    if (await _recorder.isRecording()) {
      await _recorder.stop();
    }

    setState(() {
      _isRecording = false;
      _isAnalyzing = false;
      _savedCurrent = false;
      _seconds = 0;
      _error = null;
      _recordingPath = null;
      _startedAt = null;
      _fileSizeBytes = 0;
      _avgLiveDb = 0;
      _peakLiveDb = -160;
      _dbSum = 0;
      _dbSamples = 0;
      _lastAmplitudeAt = null;
    });

    try {
      if (!await _recorder.hasPermission()) {
        if (!mounted) return;
        setState(() => _error = 'ยังไม่ได้อนุญาตไมโครโฟน');
        return;
      }

      final dir = await getApplicationDocumentsDirectory();
      final stamp = DateTime.now()
          .toIso8601String()
          .replaceAll(':', '-')
          .replaceAll('.', '-');
      final path = '${dir.path}/lifecheck_v083_$stamp.wav';

      await _recorder.start(
        const RecordConfig(
          encoder: AudioEncoder.wav,
          sampleRate: 16000,
          bitRate: 128000,
          numChannels: 1,
        ),
        path: path,
      );

      if (!mounted) return;

      setState(() {
        _recordingPath = path;
        _startedAt = DateTime.now();
        _isRecording = true;
      });

      _timer = Timer.periodic(const Duration(seconds: 1), (_) {
        if (mounted) setState(() => _seconds++);
      });

      _amplitudeSub = _recorder
          .onAmplitudeChanged(const Duration(milliseconds: 500))
          .listen(_handleAmplitude, onError: (_) {});

      _meterTimer = Timer.periodic(const Duration(milliseconds: 500), (_) async {
        if (!mounted || !_isRecording) return;
        try {
          final amplitude = await _recorder.getAmplitude();
          _handleAmplitude(amplitude);
        } catch (_) {}
      });
    } catch (e) {
      if (mounted) setState(() => _error = 'Recording error: $e');
    }
  }

  void _handleAmplitude(Amplitude amplitude) {
    if (!_isRecording || !mounted) return;

    final candidates = <double>[amplitude.current, amplitude.max]
        .where((v) => !v.isNaN && !v.isInfinite && v > -120)
        .toList();

    if (candidates.isEmpty) return;

    final currentDb = candidates.reduce(math.max);
    final now = DateTime.now();

    if (_lastAmplitudeAt != null &&
        now.difference(_lastAmplitudeAt!).inMilliseconds < 200) {
      return;
    }

    _lastAmplitudeAt = now;

    setState(() {
      _dbSamples++;
      _dbSum += currentDb;
      _avgLiveDb = _dbSum / _dbSamples;
      if (currentDb > _peakLiveDb) _peakLiveDb = currentDb;
    });
  }

  Future<int> _resolveFileSizeWithRetry(String path) async {
    final file = File(path);
    int lastSize = 0;

    for (var attempt = 0; attempt < 10; attempt++) {
      try {
        if (await file.exists()) {
          final size = await file.length();
          lastSize = size;
          if (size > 0) return size;
        }
      } catch (_) {}
      await Future.delayed(const Duration(milliseconds: 150));
    }

    return lastSize;
  }

  Future<void> _stopRecording() async {
    if (!_isRecording || _savedCurrent) return;

    _timer?.cancel();
    _meterTimer?.cancel();
    await _amplitudeSub?.cancel();

    setState(() {
      _isRecording = false;
      _isAnalyzing = true;
    });

    try {
      final path = await _recorder.stop() ?? _recordingPath;
      final started = _startedAt;

      if (path == null || started == null) {
        if (!mounted) return;
        setState(() {
          _isAnalyzing = false;
          _error = 'No recording path';
        });
        return;
      }

      final fileSize = await _resolveFileSizeWithRetry(path);
      final result = _analyzeWavStreaming(path, _seconds);

      final segments = result.segments;
      final session = SleepRecordingSession(
        startedAt: started,
        endedAt: DateTime.now(),
        durationSeconds: _seconds,
        audioPath: path,
        fileSizeBytes: fileSize,
        speech: segments.where((e) => e.type == SoundType.speech).length,
        possibleSnore:
            segments.where((e) => e.type == SoundType.possibleSnore).length,
        impact: segments.where((e) => e.type == SoundType.impact).length,
        unknown: segments.where((e) => e.type == SoundType.unknown).length,
        avgDbfs: result.avgDbfs,
        peakDbfs: result.peakDbfs,
        analysisNote: result.note,
        graphPoints: result.graphPoints,
        segments: segments,
      );

      await SleepSessionStore.append(session);

      if (!mounted) return;

      setState(() {
        _isAnalyzing = false;
        _savedCurrent = true;
        _fileSizeBytes = fileSize;
      });

      widget.onSessionSaved(session);
    } catch (e) {
      if (mounted) {
        setState(() {
          _isAnalyzing = false;
          _error = 'Stop/analyze error: $e';
        });
      }
    }
  }

  WavInfo? _readWavInfo(RandomAccessFile raf, int fileLen) {
    Uint8List readAt(int position, int length) {
      raf.setPositionSync(position);
      return raf.readSync(length);
    }

    String fourCc(Uint8List bytes, int offset) {
      if (offset + 4 > bytes.length) return '';
      return String.fromCharCodes(bytes.sublist(offset, offset + 4));
    }

    if (fileLen < 44) return null;

    final riff = readAt(0, 12);
    if (riff.length < 12 || fourCc(riff, 0) != 'RIFF' || fourCc(riff, 8) != 'WAVE') {
      return null;
    }

    var offset = 12;
    int? audioFormat;
    int? channels;
    int? sampleRate;
    int? bitsPerSample;
    int? dataStart;
    int? dataSize;
    int? fmtDataStart;
    int? fmtSize;

    while (offset + 8 <= fileLen) {
      final header = readAt(offset, 8);
      if (header.length < 8) break;

      final id = fourCc(header, 0);
      final d = ByteData.sublistView(header);
      final size = d.getUint32(4, Endian.little);
      final start = offset + 8;

      if (start + size > fileLen) {
        if (id == 'data') {
          dataStart = start;
          dataSize = math.max(0, fileLen - start);
          break;
        }
        return null;
      }

      if (id == 'fmt ') {
        fmtDataStart = start;
        fmtSize = size;
        if (size < 16) return null;

        final fmtBytes = readAt(start, math.min(size, 64));
        if (fmtBytes.length < 16) return null;

        final fd = ByteData.sublistView(fmtBytes);
        audioFormat = fd.getUint16(0, Endian.little);
        channels = fd.getUint16(2, Endian.little);
        sampleRate = fd.getUint32(4, Endian.little);
        bitsPerSample = fd.getUint16(14, Endian.little);
      } else if (id == 'data') {
        dataStart = start;
        dataSize = size;
        if (start + size > fileLen || size == 0) {
          dataSize = math.max(0, fileLen - start);
        }
        break;
      }

      offset = start + size + (size.isOdd ? 1 : 0);
    }

    var effectiveFormat = audioFormat;

    if (audioFormat == 0xFFFE &&
        fmtDataStart != null &&
        fmtSize != null &&
        fmtSize >= 40) {
      final sf = readAt(fmtDataStart + 24, 2);
      if (sf.length >= 2) {
        effectiveFormat = ByteData.sublistView(sf).getUint16(0, Endian.little);
      }
    }

    final isFloat = effectiveFormat == 3;
    final isPcm = effectiveFormat == 1;
    final bitsOk = bitsPerSample == 8 ||
        bitsPerSample == 16 ||
        bitsPerSample == 24 ||
        bitsPerSample == 32;

    if (!(isPcm || isFloat) ||
        channels == null ||
        channels <= 0 ||
        sampleRate == null ||
        sampleRate <= 0 ||
        bitsPerSample == null ||
        !bitsOk ||
        dataStart == null ||
        dataSize == null ||
        dataSize <= 0) {
      return null;
    }

    return WavInfo(
      sampleRate: sampleRate,
      channels: channels,
      bitsPerSample: bitsPerSample,
      dataStart: dataStart,
      dataSize: dataSize,
      bytesPerFrame: channels * (bitsPerSample ~/ 8),
      isFloat: isFloat,
    );
  }

  AudioAnalysisResult _analyzeWavStreaming(String path, int durationSec) {
    RandomAccessFile? raf;

    try {
      final file = File(path);
      if (!file.existsSync()) {
        return const AudioAnalysisResult(
          readable: false,
          avgDbfs: -160,
          peakDbfs: -160,
          note: 'file_not_found',
          graphPoints: [],
          segments: [],
        );
      }

      raf = file.openSync(mode: FileMode.read);
      final fileLen = raf.lengthSync();
      final info = _readWavInfo(raf, fileLen);

      if (info == null) {
        return const AudioAnalysisResult(
          readable: false,
          avgDbfs: -160,
          peakDbfs: -160,
          note: 'unsupported_wav_or_bad_header',
          graphPoints: [],
          segments: [],
        );
      }

      final int bits = info.bitsPerSample;
      final int channels = info.channels;
      final int bytesPerSample = bits ~/ 8;
      final int bytesPerFrame = info.bytesPerFrame;
      final int totalFrames = info.dataSize ~/ bytesPerFrame;

      if (totalFrames <= 0) {
        return const AudioAnalysisResult(
          readable: false,
          avgDbfs: -160,
          peakDbfs: -160,
          note: 'no_frames',
          graphPoints: [],
          segments: [],
        );
      }

      double readSampleNorm(ByteData d, int byteOffset) {
        switch (bits) {
          case 8:
            return (d.getUint8(byteOffset) - 128) / 128.0;
          case 16:
            return d.getInt16(byteOffset, Endian.little) / 32768.0;
          case 24:
            final b0 = d.getUint8(byteOffset);
            final b1 = d.getUint8(byteOffset + 1);
            final b2 = d.getUint8(byteOffset + 2);
            var v = b0 | (b1 << 8) | (b2 << 16);
            if ((v & 0x800000) != 0) v -= 0x1000000;
            return v / 8388608.0;
          case 32:
            if (info.isFloat) return d.getFloat32(byteOffset, Endian.little);
            return d.getInt32(byteOffset, Endian.little) / 2147483648.0;
          default:
            return 0.0;
        }
      }

      double toDb(double linear) {
        return linear <= 0 ? -160.0 : 20 * math.log(linear) / math.ln10;
      }

      double dbToLin(double db) => math.pow(10, db / 20).toDouble();

      const blockMs = 100;
      final framesPerBlock = math.max(1, info.sampleRate * blockMs ~/ 1000);
      final stride = math.max(1, info.sampleRate ~/ 8000);
      final blockCount = math.max(1, (totalFrames / framesPerBlock).ceil());
      final blockSec = blockMs / 1000.0;

      final rms = <double>[];
      final peak = <double>[];

      raf.setPositionSync(info.dataStart);

      for (var b = 0; b < blockCount; b++) {
        final frameStart = b * framesPerBlock;
        final framesToProcess = math.min(framesPerBlock, totalFrames - frameStart);

        if (framesToProcess <= 0) {
          rms.add(-160);
          peak.add(-160);
          continue;
        }

        final bytesToRead = framesToProcess * bytesPerFrame;
        final buffer = raf.readSync(bytesToRead);
        final actualFrames = buffer.length ~/ bytesPerFrame;

        if (actualFrames <= 0) {
          rms.add(-160);
          peak.add(-160);
          continue;
        }

        final d = ByteData.sublistView(buffer);
        double sumSq = 0;
        double maxAbs = 0;
        int count = 0;

        for (var frame = 0; frame < actualFrames; frame += stride) {
          final frameOffset = frame * bytesPerFrame;
          if (frameOffset + bytesPerFrame > buffer.length) break;

          double loudest = 0;
          for (var ch = 0; ch < channels; ch++) {
            final v = readSampleNorm(
              d,
              frameOffset + (ch * bytesPerSample),
            ).abs();
            if (v > loudest) loudest = v;
          }

          if (loudest > maxAbs) maxAbs = loudest;
          sumSq += loudest * loudest;
          count++;
        }

        if (count <= 0) {
          rms.add(-160);
          peak.add(-160);
        } else {
          rms.add(toDb(math.sqrt(sumSq / count)));
          peak.add(toDb(maxAbs));
        }
      }

      final finite = rms.where((e) => e > -120).toList()..sort();
      final avg = finite.isEmpty
          ? -160.0
          : finite.reduce((a, b) => a + b) / finite.length;
      final pk = peak.isEmpty ? -160.0 : peak.reduce(math.max);
      final maxBlockRms = rms.isEmpty ? -160.0 : rms.reduce(math.max);

      if (finite.isEmpty) {
        return AudioAnalysisResult(
          readable: true,
          avgDbfs: -160,
          peakDbfs: pk,
          note: 'v083 silent_or_zero_pcm blocks=$blockCount events=0',
          graphPoints: _makeGraphPoints(rms, peak, const [], durationSec),
          segments: const [],
        );
      }

      double percentile(List<double> sorted, double p) {
        if (sorted.isEmpty) return -160.0;
        final idx = ((sorted.length - 1) * p).round().clamp(0, sorted.length - 1);
        return sorted[idx];
      }

      const usableGateDb = -74.0;
      final floor = percentile(finite, 0.15);
      final p95 = percentile(finite, 0.95);
      final span = (p95 - floor).clamp(0.0, 60.0);
      final onsetMargin = (span * 0.26).clamp(3.5, 9.0);
      final activation = math.max(usableGateDb, floor + onsetMargin);

      final rawSegments = <SleepEventSegment>[];
      var id = 1;
      var i = 0;

      while (i < blockCount) {
        // Ignore startup transient in the first half second.
        if (i * blockSec < 0.5) {
          i++;
          continue;
        }

        if (rms[i] < activation) {
          i++;
          continue;
        }

        final startBlock = i;
        var endBlock = i;
        var gap = 0;

        var k = i + 1;
        while (k < blockCount) {
          if (rms[k] >= activation - 1.5) {
            endBlock = k;
            gap = 0;
          } else if (gap < 3) {
            gap++;
          } else {
            break;
          }
          k++;
        }

        var segPeak = -160.0;
        double segEnergy = 0;
        var segCount = 0;

        for (var b = startBlock; b <= endBlock; b++) {
          if (peak[b] > segPeak) segPeak = peak[b];
          final lin = dbToLin(rms[b]);
          segEnergy += lin * lin;
          segCount++;
        }

        final segAvg = segCount <= 0 ? -160.0 : toDb(math.sqrt(segEnergy / segCount));
        final dur = math.max(blockSec, (endBlock - startBlock + 1) * blockSec);
        final crest = segPeak - segAvg;
        final prominence = segPeak - floor;

        final type = _classifyStable(
          avgDbfs: segAvg,
          peakDbfs: segPeak,
          durationSec: dur,
          crestDb: crest,
          prominenceDb: prominence,
        );

        rawSegments.add(
          SleepEventSegment(
            id: id++,
            startSeconds: (startBlock * blockSec).floor(),
            endSeconds: math.max(
              (startBlock * blockSec).floor() + 1,
              ((endBlock + 1) * blockSec).ceil(),
            ),
            type: type,
            avgDbfs: segAvg,
            peakDbfs: segPeak,
            detail:
                'v083 floor=${floor.toStringAsFixed(1)} activation=${activation.toStringAsFixed(1)} '
                'avg=${segAvg.toStringAsFixed(1)} peak=${segPeak.toStringAsFixed(1)} '
                'crest=${crest.toStringAsFixed(1)} prom=${prominence.toStringAsFixed(1)} '
                'duration=${dur.toStringAsFixed(1)}',
          ),
        );

        i = math.max(k, endBlock + 1);
      }

      final segments = _mergeSegments(rawSegments);
      final points = _makeGraphPoints(rms, peak, segments, durationSec);

      return AudioAnalysisResult(
        readable: true,
        avgDbfs: avg,
        peakDbfs: pk,
        note:
            'v083 streaming floor=${floor.toStringAsFixed(1)}dBFS '
            'activation=${activation.toStringAsFixed(1)}dBFS '
            'maxBlockRms=${maxBlockRms.toStringAsFixed(1)}dBFS '
            'rawEvents=${rawSegments.length} mergedEvents=${segments.length}',
        graphPoints: points,
        segments: segments,
      );
    } catch (e) {
      return AudioAnalysisResult(
        readable: false,
        avgDbfs: -160,
        peakDbfs: -160,
        note: 'v083 exception=$e',
        graphPoints: const [],
        segments: const [],
      );
    } finally {
      raf?.closeSync();
    }
  }

  SoundType _classifyStable({
    required double avgDbfs,
    required double peakDbfs,
    required double durationSec,
    required double crestDb,
    required double prominenceDb,
  }) {
    // Short, sharp sounds are usually impact/knock.
    if (durationSec <= 0.80 &&
        (peakDbfs >= -24 || crestDb >= 8 || prominenceDb >= 12)) {
      return SoundType.impact;
    }

    // Possible snore (still conservative, but less strict than v0.8.2).
    if (durationSec >= 0.8 &&
        durationSec <= 12.0 &&
        avgDbfs >= -68 &&
        avgDbfs <= -18 &&
        peakDbfs <= -4 &&
        crestDb <= 18 &&
        prominenceDb >= 4.5) {
      return SoundType.possibleSnore;
    }

    // Speech-like: allow softer levels to avoid missing real speech.
    if (durationSec >= 0.20 &&
        durationSec <= 18.0 &&
        avgDbfs >= -72 &&
        prominenceDb >= 3.0) {
      return SoundType.speech;
    }

    return SoundType.unknown;
  }

  List<SleepEventSegment> _mergeSegments(List<SleepEventSegment> input) {
    if (input.isEmpty) return const [];

    final sorted = [...input]..sort((a, b) => a.startSeconds.compareTo(b.startSeconds));
    final merged = <SleepEventSegment>[];

    SleepEventSegment current = sorted.first;

    for (var idx = 1; idx < sorted.length; idx++) {
      final next = sorted[idx];
      final gap = next.startSeconds - current.endSeconds;

      if (current.type == next.type && gap <= 3) {
        final durationA = current.durationSeconds;
        final durationB = next.durationSeconds;
        final totalDuration = math.max(1, durationA + durationB);
        final avg = ((current.avgDbfs * durationA) + (next.avgDbfs * durationB)) / totalDuration;
        final peak = math.max(current.peakDbfs, next.peakDbfs);

        current = current.copyWith(
          endSeconds: math.max(current.endSeconds, next.endSeconds),
          avgDbfs: avg,
          peakDbfs: peak,
          detail: '${current.detail} · merged_with=${next.id}',
        );
      } else {
        merged.add(current.copyWith(id: merged.length + 1));
        current = next;
      }
    }

    merged.add(current.copyWith(id: merged.length + 1));
    return merged;
  }

  List<SleepGraphPoint> _makeGraphPoints(
    List<double> rms,
    List<double> peak,
    List<SleepEventSegment> segments,
    int durationSec,
  ) {
    if (rms.isEmpty || peak.isEmpty) return const [];

    final points = <SleepGraphPoint>[];
    const blocksPerPoint = 10; // 1 second because block = 100ms

    for (int b = 0; b < rms.length; b += blocksPerPoint) {
      final end = math.min(rms.length, b + blocksPerPoint);

      double sumDb = 0;
      double maxPeak = -160;
      int count = 0;

      for (int k = b; k < end; k++) {
        sumDb += rms[k];
        if (peak[k] > maxPeak) maxPeak = peak[k];
        count++;
      }

      final sec = (b * 0.1).floor();

      points.add(
        SleepGraphPoint(
          offsetSeconds: sec,
          avgDbfs: count <= 0 ? -160 : sumDb / count,
          peakDbfs: maxPeak,
          type: _typeAt(sec, segments),
        ),
      );
    }

    return points;
  }

  SoundType _typeAt(int sec, List<SleepEventSegment> segments) {
    for (final s in segments) {
      if (sec >= s.startSeconds && sec <= s.endSeconds) {
        return s.type;
      }
    }
    return SoundType.background;
  }

class TimelinePainter extends CustomPainter {
  final List<SleepGraphPoint> points;
  final List<SleepEventSegment> segments;
  final int durationSeconds;
  final bool showDemoWhenEmpty;

  TimelinePainter({
    required this.points,
    required this.segments,
    required this.durationSeconds,
    required this.showDemoWhenEmpty,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final bg = RRect.fromRectAndRadius(
      Offset.zero & size,
      const Radius.circular(16),
    );

    canvas.drawRRect(
      bg,
      Paint()..color = AppColors.navy2.withOpacity(0.92),
    );

    final left = 30.0;
    final right = size.width - 10;
    final top = 13.0;
    final bottom = size.height - 24;
    final width = math.max(1.0, right - left);
    final height = math.max(1.0, bottom - top);

    final gridPaint = Paint()
      ..color = AppColors.white.withOpacity(0.08)
      ..strokeWidth = 1;

    for (final db in [-80, -60, -40, -20, 0]) {
      final y = _dbToY(db.toDouble(), top, height);
      canvas.drawLine(Offset(left, y), Offset(right, y), gridPaint);
      _drawText(
        canvas,
        '$db',
        Offset(6, y - 6),
        AppColors.muted.withOpacity(0.70),
        9,
      );
    }

    final data = points.isEmpty && showDemoWhenEmpty
        ? _demo(durationSeconds <= 0 ? 60 : durationSeconds)
        : points;

    if (data.isEmpty) {
      _drawText(
        canvas,
        'No graph data yet',
        Offset(left + 14, top + 18),
        AppColors.muted,
        12,
      );
      return;
    }

    final dur = math.max(1, durationSeconds <= 0 ? data.last.offsetSeconds : durationSeconds);
    final barWidth = math.max(2.0, width / math.max(1, data.length) * 0.76);

    for (final point in data) {
      final x = left + (point.offsetSeconds / dur) * width;
      final y = _dbToY(point.peakDbfs, top, height);
      final c = colorForType(point.type).withOpacity(
        point.type == SoundType.background ? 0.68 : 0.98,
      );
      final topY = math.min(bottom - 3, y);

      canvas.drawRRect(
        RRect.fromRectAndRadius(
          Rect.fromLTWH(x, topY, barWidth, bottom - topY),
          const Radius.circular(3),
        ),
        Paint()..color = c,
      );
    }

    for (final segment in segments) {
      final x1 = left + (segment.startSeconds / dur) * width;
      final x2 = left + (segment.endSeconds / dur) * width;
      canvas.drawRRect(
        RRect.fromRectAndRadius(
          Rect.fromLTRB(x1, top + 1, math.max(x1 + 4, x2), top + 8),
          const Radius.circular(4),
        ),
        Paint()..color = colorForType(segment.type),
      );
    }

    _drawText(canvas, 'Time →', Offset(right - 48, bottom + 5), AppColors.muted, 10);
    _drawText(canvas, 'dBFS', Offset(4, top - 4), AppColors.muted, 9);
  }

  List<SleepGraphPoint> _demo(int duration) {
    final out = <SleepGraphPoint>[];

    for (int i = 0; i <= duration; i += 2) {
      var db = -76.0 + math.sin(i * 0.55) * 2;
      var type = SoundType.background;

      if (i >= duration * 0.18 && i <= duration * 0.30) {
        db = -34 + math.sin(i).abs() * 8;
        type = SoundType.speech;
      } else if (i >= duration * 0.44 && i <= duration * 0.58) {
        db = -42 + math.sin(i * 0.7).abs() * 7;
        type = SoundType.possibleSnore;
      } else if (i >= duration * 0.66 && i <= duration * 0.70) {
        db = -10;
        type = SoundType.impact;
      } else if (i >= duration * 0.80 && i <= duration * 0.90) {
        db = -47 + math.sin(i * 0.8).abs() * 7;
        type = SoundType.unknown;
      }

      out.add(
        SleepGraphPoint(
          offsetSeconds: i,
          avgDbfs: db - 3,
          peakDbfs: db,
          type: type,
        ),
      );
    }

    return out;
  }

  double _dbToY(double db, double top, double height) {
    final clamped = db.clamp(-80.0, 0.0).toDouble();
    final normalized = (clamped + 80.0) / 80.0;
    return top + height * (1 - normalized);
  }

  void _drawText(Canvas canvas, String text, Offset offset, Color color, double size) {
    final painter = TextPainter(
      text: TextSpan(
        text: text,
        style: TextStyle(color: color, fontSize: size),
      ),
      textDirection: TextDirection.ltr,
    )..layout();

    painter.paint(canvas, offset);
  }

  @override
  bool shouldRepaint(covariant TimelinePainter oldDelegate) {
    return true;
  }
}

class FakeWaveformPainter extends CustomPainter {
  final bool active;

  FakeWaveformPainter({required this.active});

  @override
  void paint(Canvas canvas, Size size) {
    final p = Paint()
      ..color = (active ? AppColors.cyan : AppColors.greyEvent).withOpacity(0.76)
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;

    final center = size.height / 2;
    final count = 54;

    for (int i = 0; i < count; i++) {
      final x = (i / (count - 1)) * size.width;
      final amp = active
          ? 4 + math.sin(i * 0.55).abs() * 18 + math.sin(i * 1.15).abs() * 8
          : 3 + math.sin(i * 0.4).abs() * 4;
      canvas.drawLine(Offset(x, center - amp), Offset(x, center + amp), p);
    }
  }

  @override
  bool shouldRepaint(covariant FakeWaveformPainter oldDelegate) {
    return oldDelegate.active != active;
  }
}

String formatDuration(int seconds) {
  final h = seconds ~/ 3600;
  final m = (seconds % 3600) ~/ 60;
  final s = seconds % 60;

  if (h > 0) {
    return '${h.toString().padLeft(2, '0')}:'
        '${m.toString().padLeft(2, '0')}:'
        '${s.toString().padLeft(2, '0')}';
  }

  return '${m.toString().padLeft(2, '0')}:'
      '${s.toString().padLeft(2, '0')}';
}
