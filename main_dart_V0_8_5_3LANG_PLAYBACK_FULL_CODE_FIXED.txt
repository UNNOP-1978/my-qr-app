String soundTypeToJson(SoundType type) {
  switch (type) {
    case SoundType.speech:
      return 'speech';
    case SoundType.possibleSnore:
      return 'possible_snore';
    case SoundType.unknown:
      return 'unknown';
    case SoundType.impact:
      return 'impact';
    case SoundType.background:
      return 'background';
  }
}

SoundType soundTypeFromJson(dynamic value) {
  final v = value?.toString() ?? '';
  if (v == 'speech') return SoundType.speech;
  if (v == 'possible_snore' || v == 'snore') return SoundType.possibleSnore;
  if (v == 'unknown') return SoundType.unknown;
  if (v == 'impact') return SoundType.impact;
  return SoundType.background;
}

class JsonTools {
  static int i(dynamic value) {
    if (value is int) return value;
    if (value is num) return value.toInt();
    return int.tryParse(value?.toString() ?? '') ?? 0;
  }

  static double d(dynamic value) {
    if (value is double) return value;
    if (value is num) return value.toDouble();
    return double.tryParse(value?.toString() ?? '') ?? 0.0;
  }
}

class SleepGraphPoint {
  final int offsetSeconds;
  final double avgDbfs;
  final double peakDbfs;
  final SoundType type;

  const SleepGraphPoint({
    required this.offsetSeconds,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.type,
  });

  Map<String, dynamic> toJson() => {
        'offsetSeconds': offsetSeconds,
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'type': soundTypeToJson(type),
      };

  factory SleepGraphPoint.fromJson(Map<String, dynamic> json) {
    return SleepGraphPoint(
      offsetSeconds: JsonTools.i(json['offsetSeconds']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      type: soundTypeFromJson(json['type']),
    );
  }
}

class SleepEventSegment {
  final int id;
  final int startSeconds;
  final int endSeconds;
  final SoundType type;
  final double avgDbfs;
  final double peakDbfs;
  final double crestDb;
  final double stdDb;
  final String detail;

  const SleepEventSegment({
    required this.id,
    required this.startSeconds,
    required this.endSeconds,
    required this.type,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.crestDb,
    required this.stdDb,
    required this.detail,
  });

  int get durationSeconds => math.max(1, endSeconds - startSeconds);

  Map<String, dynamic> toJson() => {
        'id': id,
        'startSeconds': startSeconds,
        'endSeconds': endSeconds,
        'type': soundTypeToJson(type),
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'crestDb': crestDb,
        'stdDb': stdDb,
        'detail': detail,
      };

  factory SleepEventSegment.fromJson(Map<String, dynamic> json) {
    return SleepEventSegment(
      id: JsonTools.i(json['id']),
      startSeconds: JsonTools.i(json['startSeconds']),
      endSeconds: JsonTools.i(json['endSeconds']),
      type: soundTypeFromJson(json['type']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      crestDb: JsonTools.d(json['crestDb']),
      stdDb: JsonTools.d(json['stdDb']),
      detail: json['detail']?.toString() ?? '',
    );
  }
}

class SleepRecordingSession {
  final DateTime startedAt;
  final DateTime endedAt;
  final int durationSeconds;
  final String audioPath;
  final int speech;
  final int possibleSnore;
  final int impact;
  final int unknown;
  final double avgDbfs;
  final double peakDbfs;
  final String analysisNote;
  final List<SleepGraphPoint> graphPoints;
  final List<SleepEventSegment> segments;

  const SleepRecordingSession({
    required this.startedAt,
    required this.endedAt,
    required this.durationSeconds,
    required this.audioPath,
    required this.speech,
    required this.possibleSnore,
    required this.impact,
    required this.unknown,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.analysisNote,
    required this.graphPoints,
    required this.segments,
  });

  Map<String, dynamic> toJson() => {
        'version': kAppVersion,
        'startedAt': startedAt.toIso8601String(),
        'endedAt': endedAt.toIso8601String(),
        'durationSeconds': durationSeconds,
        'audioPath': audioPath,
        'speech': speech,
        'possibleSnore': possibleSnore,
        'impact': impact,
        'unknown': unknown,
        'avgDbfs': avgDbfs,
        'peakDbfs': peakDbfs,
        'analysisNote': analysisNote,
        'graphPoints': graphPoints.map((e) => e.toJson()).toList(),
        'segments': segments.map((e) => e.toJson()).toList(),
      };

  factory SleepRecordingSession.fromJson(Map<String, dynamic> json) {
    final gp = json['graphPoints'];
    final seg = json['segments'];
    return SleepRecordingSession(
      startedAt: DateTime.tryParse(json['startedAt']?.toString() ?? '') ?? DateTime.now(),
      endedAt: DateTime.tryParse(json['endedAt']?.toString() ?? '') ?? DateTime.now(),
      durationSeconds: JsonTools.i(json['durationSeconds']),
      audioPath: json['audioPath']?.toString() ?? '',
      speech: JsonTools.i(json['speech']),
      possibleSnore: JsonTools.i(json['possibleSnore']),
      impact: JsonTools.i(json['impact']),
      unknown: JsonTools.i(json['unknown']),
      avgDbfs: JsonTools.d(json['avgDbfs']),
      peakDbfs: JsonTools.d(json['peakDbfs']),
      analysisNote: json['analysisNote']?.toString() ?? '',
      graphPoints: gp is List
          ? gp
              .whereType<Map>()
              .map((e) => SleepGraphPoint.fromJson(e.map((k, v) => MapEntry(k.toString(), v))))
              .toList()
          : const [],
      segments: seg is List
          ? seg
              .whereType<Map>()
              .map((e) => SleepEventSegment.fromJson(e.map((k, v) => MapEntry(k.toString(), v))))
              .toList()
          : const [],
    );
  }

  String get durationText => formatDuration(durationSeconds);

  bool get hasAudioFile => audioPath.isNotEmpty && File(audioPath).existsSync();

  String get audioFileName {
    if (audioPath.isEmpty) return '-';
    return audioPath.split(Platform.pathSeparator).last;
  }

  String get rangeText {
    String t(DateTime d) => '${d.hour.toString().padLeft(2, '0')}:${d.minute.toString().padLeft(2, '0')}';
    return '${t(startedAt)} - ${t(endedAt)}';
  }
}

class SleepSessionStore {
  static const _fileName = 'lifecheck_sleep_sessions_v085_3lang_playback.json';

  static Future<File> _file() async {
    final d = await getApplicationDocumentsDirectory();
    return File('${d.path}/$_fileName');
  }

  static Future<SleepRecordingSession?> loadLatest() async {
    try {
      final f = await _file();
      if (!await f.exists()) return null;
      final raw = await f.readAsString();
      if (raw.trim().isEmpty) return null;
      final decoded = jsonDecode(raw);
      if (decoded is! List || decoded.isEmpty) return null;
      final last = decoded.last;
      if (last is! Map) return null;
      return SleepRecordingSession.fromJson(last.map((k, v) => MapEntry(k.toString(), v)));
    } catch (_) {
      return null;
    }
  }

  static Future<void> append(SleepRecordingSession session) async {
    final f = await _file();
    List<dynamic> decoded = [];
    if (await f.exists()) {
      try {
        final raw = await f.readAsString();
        final item = jsonDecode(raw);
        if (item is List) decoded = item;
      } catch (_) {}
    }
    decoded.add(session.toJson());
    await f.writeAsString(const JsonEncoder.withIndent('  ').convert(decoded), flush: true);
  }

  static Future<void> clearAll() async {
    final f = await _file();
    if (await f.exists()) {
      await f.delete();
    }
  }
}

class LanguageStore {
  static const _fileName = 'lifecheck_sleep_language_v085.txt';

  static Future<File> _file() async {
    final dir = await getApplicationDocumentsDirectory();
    return File('${dir.path}/$_fileName');
  }

  static Future<AppLanguage> load() async {
    try {
      final file = await _file();
      if (!await file.exists()) return AppLanguage.en;
      final raw = (await file.readAsString()).trim();
      if (raw == 'th') return AppLanguage.th;
      if (raw == 'ja') return AppLanguage.ja;
      return AppLanguage.en;
    } catch (_) {
      return AppLanguage.en;
    }
  }

  static Future<void> save(AppLanguage language) async {
    final file = await _file();
    var value = 'en';
    if (language == AppLanguage.th) value = 'th';
    if (language == AppLanguage.ja) value = 'ja';
    await file.writeAsString(value, flush: true);
  }
}

class WavInfo {
  final int sampleRate;
  final int channels;
  final int bitsPerSample;
  final int dataStart;
  final int dataSize;
  final int bytesPerFrame;
  final bool isFloat;

  const WavInfo({
    required this.sampleRate,
    required this.channels,
    required this.bitsPerSample,
    required this.dataStart,
    required this.dataSize,
    required this.bytesPerFrame,
    required this.isFloat,
  });
}

class AudioAnalysisResult {
  final bool readable;
  final double avgDbfs;
  final double peakDbfs;
  final String note;
  final List<SleepGraphPoint> graphPoints;
  final List<SleepEventSegment> segments;

  const AudioAnalysisResult({
    required this.readable,
    required this.avgDbfs,
    required this.peakDbfs,
    required this.note,
    required this.graphPoints,
    required this.segments,
  });
}

class WavAnalyzer {
  static AudioAnalysisResult analyze({
    required String path,
    required int durationSeconds,
  }) {
    RandomAccessFile? raf;
    try {
      final file = File(path);
      if (!file.existsSync()) {
        return _fail('v085 file_not_found path=$path');
      }
      raf = file.openSync(mode: FileMode.read);
      final fileLen = raf.lengthSync();
      if (fileLen < 44) {
        return _fail('v085 too_short bytes=$fileLen');
      }
      final info = _readWavInfo(raf, fileLen);
      if (info == null) {
        return _fail('v085 unsupported_wav_or_bad_header bytes=$fileLen');
      }

      double toDb(double linear) => linear <= 0 ? -160.0 : 20 * math.log(linear) / math.ln10;
      double dbToLin(double db) => math.pow(10, db / 20).toDouble();

      const blockMs = 100;
      final framesPerBlock = math.max(1, info.sampleRate * blockMs ~/ 1000);
      final totalFrames = info.dataSize ~/ info.bytesPerFrame;
      final blockCount = math.max(1, (totalFrames / framesPerBlock).ceil());
      final stride = math.max(1, info.sampleRate ~/ 8000);

      final rmsDb = <double>[];
      final peakDb = <double>[];

      raf.setPositionSync(info.dataStart);
      for (var b = 0; b < blockCount; b++) {
        final frameStart = b * framesPerBlock;
        final framesThisBlock = math.min(framesPerBlock, totalFrames - frameStart);
        if (framesThisBlock <= 0) {
          rmsDb.add(-160);
          peakDb.add(-160);
          continue;
        }

        final bytesToRead = framesThisBlock * info.bytesPerFrame;
        final buffer = raf.readSync(bytesToRead);
        if (buffer.isEmpty) {
          rmsDb.add(-160);
          peakDb.add(-160);
          continue;
        }
        final data = ByteData.sublistView(buffer);
        final actualFrames = buffer.length ~/ info.bytesPerFrame;

        double sumSq = 0;
        double maxAbs = 0;
        int count = 0;
        for (var frame = 0; frame < actualFrames; frame += stride) {
          final base = frame * info.bytesPerFrame;
          if (base + info.bytesPerFrame > buffer.length) break;
          double loudest = 0;
          for (var ch = 0; ch < info.channels; ch++) {
            final sampleOffset = base + ch * (info.bitsPerSample ~/ 8);
            final v = _readSample(data, sampleOffset, info).abs();
            if (v > loudest) loudest = v;
          }
          sumSq += loudest * loudest;
          if (loudest > maxAbs) maxAbs = loudest;
          count++;
        }
        if (count <= 0) {
          rmsDb.add(-160);
          peakDb.add(-160);
        } else {
          rmsDb.add(toDb(math.sqrt(sumSq / count)));
          peakDb.add(toDb(maxAbs));
        }
      }

      final finite = rmsDb.where((e) => e > -120).toList()..sort();
      final overallAvg = finite.isEmpty ? -160.0 : finite.reduce((a, b) => a + b) / finite.length;
      final overallPeak = peakDb.isEmpty ? -160.0 : peakDb.reduce(math.max);
      final maxBlockRms = rmsDb.isEmpty ? -160.0 : rmsDb.reduce(math.max);

      if (finite.isEmpty) {
        final graph = _buildGraphPoints(rmsDb, peakDb, const []);
        return AudioAnalysisResult(
          readable: true,
          avgDbfs: -160,
          peakDbfs: overallPeak,
          note: 'v085 silent_or_zero_pcm blocks=$blockCount events=0 graph=${graph.length}',
          graphPoints: graph,
          segments: const [],
        );
      }

      double percentile(double p) {
        final index = ((finite.length - 1) * p).round().clamp(0, finite.length - 1).toInt();
        return finite[index];
      }

      final floor = percentile(0.20);
      final p90 = percentile(0.90);
      final dynamicRange = (p90 - floor).clamp(0.0, 50.0);
      final activation = math.max(-62.0, floor + math.max(6.0, dynamicRange * 0.35));
      const usableGate = -60.0;

      final segments = <SleepEventSegment>[];
      var i = 0;
      var id = 1;
      while (i < blockCount) {
        if (rmsDb[i] < activation) {
          i++;
          continue;
        }
        final start = i;
        var end = i;
        var gap = 0;
        while (end + 1 < blockCount) {
          final nextActive = rmsDb[end + 1] >= (activation - 1.5);
          if (nextActive) {
            gap = 0;
            end++;
          } else if (gap < 2) {
            gap++;
            end++;
          } else {
            break;
          }
        }

        final startSec = (start * 0.1).floor();
        final endSec = math.max(startSec + 1, ((end + 1) * 0.1).ceil());
        final durSec = math.max(0.1, (end - start + 1) * 0.1);

        double sumDb = 0;
        double sumSqForStd = 0;
        double sumLinSq = 0;
        double segPeak = -160;
        int n = 0;
        for (var k = start; k <= end && k < rmsDb.length; k++) {
          final r = rmsDb[k];
          sumDb += r;
          sumSqForStd += r * r;
          final lin = dbToLin(r);
          sumLinSq += lin * lin;
          if (peakDb[k] > segPeak) segPeak = peakDb[k];
          n++;
        }
        if (n <= 0) {
          i = end + 1;
          continue;
        }
        final segAvgDb = sumDb / n;
        final segRmsDb = toDb(math.sqrt(sumLinSq / n));
        final variance = math.max(0.0, sumSqForStd / n - segAvgDb * segAvgDb);
        final stdDb = math.sqrt(variance);
        final crest = segPeak - segRmsDb;

        if (segPeak < usableGate && maxBlockRms < usableGate) {
          i = end + 1;
          continue;
        }

        final type = _classify(
          avgDb: segAvgDb,
          peakDb: segPeak,
          durationSec: durSec,
          crestDb: crest,
          stdDb: stdDb,
        );

        segments.add(SleepEventSegment(
          id: id++,
          startSeconds: startSec,
          endSeconds: endSec,
          type: type,
          avgDbfs: segAvgDb,
          peakDbfs: segPeak,
          crestDb: crest,
          stdDb: stdDb,
          detail: 'v085 floor=${floor.toStringAsFixed(1)} activation=${activation.toStringAsFixed(1)} '
              'dur=${durSec.toStringAsFixed(1)} crest=${crest.toStringAsFixed(1)} std=${stdDb.toStringAsFixed(1)}',
        ));

        i = end + 1;
      }

      final merged = _mergeSegments(segments);
      final graph = _buildGraphPoints(rmsDb, peakDb, merged);

      final note = 'v085 3lang readable=true floor=${floor.toStringAsFixed(1)}dBFS '
          'p90=${p90.toStringAsFixed(1)}dBFS activation=${activation.toStringAsFixed(1)}dBFS '
          'maxRms=${maxBlockRms.toStringAsFixed(1)}dBFS peak=${overallPeak.toStringAsFixed(1)}dBFS '
          'blocks=$blockCount events=${merged.length}';

      return AudioAnalysisResult(
        readable: true,
        avgDbfs: overallAvg,
        peakDbfs: overallPeak,
        note: note,
        graphPoints: graph,
        segments: merged,
      );
    } catch (e) {
      return _fail('v085 analyzer_exception=$e');
    } finally {
      raf?.closeSync();
    }
  }

  static AudioAnalysisResult _fail(String note) {
    return AudioAnalysisResult(
      readable: false,
      avgDbfs: -160,
      peakDbfs: -160,
      note: note,
      graphPoints: const [],
      segments: const [],
    );
  }

  static WavInfo? _readWavInfo(RandomAccessFile raf, int fileLen) {
    Uint8List readAt(int pos, int len) {
      raf.setPositionSync(pos);
      return raf.readSync(len);
    }

    String four(Uint8List b, int off) {
      if (off + 4 > b.length) return '';
      return String.fromCharCodes(b.sublist(off, off + 4));
    }

    final header = readAt(0, 12);
    if (header.length < 12 || four(header, 0) != 'RIFF' || four(header, 8) != 'WAVE') {
      return null;
    }

    int? audioFormat;
    int? channels;
    int? sampleRate;
    int? bits;
    int? dataStart;
    int? dataSize;
    int? fmtStart;
    int? fmtSize;

    var offset = 12;
    while (offset + 8 <= fileLen) {
      final ch = readAt(offset, 8);
      if (ch.length < 8) break;
      final id = four(ch, 0);
      final bd = ByteData.sublistView(ch);
      final size = bd.getUint32(4, Endian.little);
      final start = offset + 8;
      if (start > fileLen) break;

      if (id == 'fmt ') {
        fmtStart = start;
        fmtSize = size;
        final fmtBytes = readAt(start, math.min(size, 64));
        if (fmtBytes.length < 16) return null;
        final f = ByteData.sublistView(fmtBytes);
        audioFormat = f.getUint16(0, Endian.little);
        channels = f.getUint16(2, Endian.little);
        sampleRate = f.getUint32(4, Endian.little);
        bits = f.getUint16(14, Endian.little);
      } else if (id == 'data') {
        dataStart = start;
        dataSize = math.min(size, math.max(0, fileLen - start));
        break;
      }

      offset = start + size + (size.isOdd ? 1 : 0);
    }

    var effectiveFormat = audioFormat;
    if (audioFormat == 0xFFFE && fmtStart != null && fmtSize != null && fmtSize >= 40) {
      final sub = readAt(fmtStart + 24, 2);
      if (sub.length >= 2) {
        effectiveFormat = ByteData.sublistView(sub).getUint16(0, Endian.little);
      }
    }

    if (effectiveFormat == null || channels == null || sampleRate == null || bits == null || dataStart == null || dataSize == null) {
      return null;
    }
    final isPcm = effectiveFormat == 1;
    final isFloat = effectiveFormat == 3;
    final bitsOk = bits == 8 || bits == 16 || bits == 24 || bits == 32;
    if (!(isPcm || isFloat) || !bitsOk || channels <= 0 || sampleRate <= 0 || dataSize <= 0) {
      return null;
    }

    return WavInfo(
      sampleRate: sampleRate,
      channels: channels,
      bitsPerSample: bits,
      dataStart: dataStart,
      dataSize: dataSize,
      bytesPerFrame: channels * (bits ~/ 8),
      isFloat: isFloat,
    );
  }

  static double _readSample(ByteData data, int byteOffset, WavInfo info) {
    switch (info.bitsPerSample) {
      case 8:
        return (data.getUint8(byteOffset) - 128) / 128.0;
      case 16:
        return data.getInt16(byteOffset, Endian.little) / 32768.0;
      case 24:
        final b0 = data.getUint8(byteOffset);
        final b1 = data.getUint8(byteOffset + 1);
        final b2 = data.getUint8(byteOffset + 2);
        var v = b0 | (b1 << 8) | (b2 << 16);
        if ((v & 0x800000) != 0) v -= 0x1000000;
        return v / 8388608.0;
      case 32:
        if (info.isFloat) return data.getFloat32(byteOffset, Endian.little).clamp(-1.0, 1.0).toDouble();
        return data.getInt32(byteOffset, Endian.little) / 2147483648.0;
      default:
        return 0.0;
    }
  }

  static SoundType _classify({
    required double avgDb,
    required double peakDb,
    required double durationSec,
    required double crestDb,
    required double stdDb,
  }) {
    // เคาะ/กระแทก: สั้นและ peak สูงมาก
    if (durationSec <= 0.55 && peakDb >= -16) return SoundType.impact;

    // Possible snore: ตั้งใจให้ไม่ strict เกินไป เพื่อให้รอบทดสอบวันนี้เห็นสีแดงได้
    // แต่ยังเรียก Possible ไม่ใช่ Confirmed
    final snoreDuration = durationSec >= 0.8 && durationSec <= 12.0;
    final snoreLevel = avgDb >= -66 && avgDb <= -6;
    final notHardImpact = crestDb <= 28;
    final fairlySteady = stdDb <= 10.5 || durationSec >= 2.0;
    if (snoreDuration && snoreLevel && notHardImpact && fairlySteady) {
      return SoundType.possibleSnore;
    }

    // Speech: ช่วงเสียงที่เหลือซึ่งดังพอและไม่สั้นมาก
    if (durationSec >= 0.30 && avgDb >= -62) return SoundType.speech;

    return SoundType.unknown;
  }

  static List<SleepEventSegment> _mergeSegments(List<SleepEventSegment> input) {
    if (input.isEmpty) return input;
    final items = [...input]..sort((a, b) => a.startSeconds.compareTo(b.startSeconds));
    final out = <SleepEventSegment>[];
    var cur = items.first;
    for (var i = 1; i < items.length; i++) {
      final next = items[i];
      final gap = next.startSeconds - cur.endSeconds;
      final sameType = next.type == cur.type;
      if (sameType && gap <= 3) {
        cur = SleepEventSegment(
          id: cur.id,
          startSeconds: cur.startSeconds,
          endSeconds: math.max(cur.endSeconds, next.endSeconds),
          type: cur.type,
          avgDbfs: (cur.avgDbfs + next.avgDbfs) / 2,
          peakDbfs: math.max(cur.peakDbfs, next.peakDbfs),
          crestDb: math.max(cur.crestDb, next.crestDb),
          stdDb: (cur.stdDb + next.stdDb) / 2,
          detail: '${cur.detail} | merged',
        );
      } else {
        out.add(cur);
        cur = next;
      }
    }
    out.add(cur);
    for (var i = 0; i < out.length; i++) {
      out[i] = SleepEventSegment(
        id: i + 1,
        startSeconds: out[i].startSeconds,
        endSeconds: out[i].endSeconds,
        type: out[i].type,
        avgDbfs: out[i].avgDbfs,
        peakDbfs: out[i].peakDbfs,
        crestDb: out[i].crestDb,
        stdDb: out[i].stdDb,
        detail: out[i].detail,
      );
    }
    return out;
  }

  static List<SleepGraphPoint> _buildGraphPoints(
    List<double> rmsDb,
    List<double> peakDb,
    List<SleepEventSegment> segments,
  ) {
    final out = <SleepGraphPoint>[];
    if (rmsDb.isEmpty) return out;

    // 1 point = 1 second, from 10 blocks of 100ms
    for (var b = 0; b < rmsDb.length; b += 10) {
      final end = math.min(rmsDb.length, b + 10);
      double sum = 0;
      double peak = -160;
      var count = 0;
      for (var k = b; k < end; k++) {
        sum += rmsDb[k];
        if (k < peakDb.length && peakDb[k] > peak) peak = peakDb[k];
        count++;
      }
      final sec = (b * 0.1).floor();
      out.add(SleepGraphPoint(
        offsetSeconds: sec,
        avgDbfs: count == 0 ? -160 : sum / count,
        peakDbfs: peak,
        type: _typeAt(sec, segments),
      ));
    }
    return out;
  }

  static SoundType _typeAt(int sec, List<SleepEventSegment> segments) {
    for (final s in segments) {
      if (sec >= s.startSeconds && sec <= s.endSeconds) return s.type;
    }
    return SoundType.background;
  }
}

class RecordingScreen extends StatefulWidget {
  final AppText t;
  final int startToken;
  final ValueChanged<SleepRecordingSession> onSessionSaved;
  final VoidCallback onViewReport;

  const RecordingScreen({
    super.key,
    required this.t,
    required this.startToken,
    required this.onSessionSaved,
    required this.onViewReport,
  });

  @override
  State<RecordingScreen> createState() => _RecordingScreenState();
}

class _RecordingScreenState extends State<RecordingScreen> {
  final AudioRecorder _recorder = AudioRecorder();
  Timer? _timer;
  StreamSubscription<Amplitude>? _amplitudeSub;
  int _seconds = 0;
  bool _isRecording = false;
  bool _isAnalyzing = false;
  String? _recordingPath;
  String? _error;
  int _lastToken = 0;
  DateTime? _startedAt;
  double _liveDb = -160;
  double _peakDb = -160;

  @override
  void didUpdateWidget(covariant RecordingScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.startToken != 0 && widget.startToken != _lastToken) {
      _lastToken = widget.startToken;
      _startRecording();
    }
  }

  Future<void> _startRecording() async {
    _timer?.cancel();
    await _amplitudeSub?.cancel();
    _amplitudeSub = null;

    try {
      if (await _recorder.isRecording()) await _recorder.stop();
    } catch (_) {}

    setState(() {
      _seconds = 0;
      _isRecording = false;
      _isAnalyzing = false;
      _recordingPath = null;
      _error = null;
      _startedAt = null;
      _liveDb = -160;
      _peakDb = -160;
    });

    try {
      final ok = await _recorder.hasPermission();
      if (!ok) {
        if (mounted) setState(() => _error = widget.t.micPermissionDenied);
        return;
      }

      final dir = await getApplicationDocumentsDirectory();
      final stamp = DateTime.now().toIso8601String().replaceAll(':', '-').replaceAll('.', '-');
      final path = '${dir.path}/lifecheck_v085_$stamp.wav';

      await _recorder.start(
        const RecordConfig(
          encoder: AudioEncoder.wav,
          sampleRate: 16000,
          bitRate: 128000,
          numChannels: 1,
        ),
        path: path,
      );

      if (!mounted) return;
      setState(() {
        _recordingPath = path;
        _startedAt = DateTime.now();
        _isRecording = true;
      });

      _timer = Timer.periodic(const Duration(seconds: 1), (_) {
        if (mounted) setState(() => _seconds++);
      });

      _amplitudeSub = _recorder.onAmplitudeChanged(const Duration(milliseconds: 500)).listen((amp) {
        if (!_isRecording || !mounted) return;
        final values = <double>[amp.current, amp.max].where((v) => !v.isNaN && !v.isInfinite && v > -120).toList();
        if (values.isEmpty) return;
        final d = values.reduce(math.max);
        setState(() {
          _liveDb = d;
          if (d > _peakDb) _peakDb = d;
        });
      });
    } catch (e) {
      if (mounted) setState(() => _error = 'Recording error: $e');
    }
  }

  Future<void> _stopRecording() async {
    if (!_isRecording) return;
    _timer?.cancel();
    await _amplitudeSub?.cancel();
    _amplitudeSub = null;

    setState(() {
      _isRecording = false;
      _isAnalyzing = true;
      _error = null;
    });

    try {
      final path = await _recorder.stop() ?? _recordingPath;
      if (path == null || _startedAt == null) {
        if (mounted) setState(() => _error = 'No recording file path');
        return;
      }

      // ให้ระบบเขียนไฟล์จบจริง ๆ ก่อนวิเคราะห์
      await Future.delayed(const Duration(milliseconds: 250));

      final result = WavAnalyzer.analyze(path: path, durationSeconds: _seconds);
      final segs = result.segments;
      final debugNote = result.note.contains('v085') ? result.note : 'v085 ${result.note}';
      final session = SleepRecordingSession(
        startedAt: _startedAt!,
        endedAt: DateTime.now(),
        durationSeconds: _seconds,
        audioPath: path,
        speech: segs.where((e) => e.type == SoundType.speech).length,
        possibleSnore: segs.where((e) => e.type == SoundType.possibleSnore).length,
        impact: segs.where((e) => e.type == SoundType.impact).length,
        unknown: segs.where((e) => e.type == SoundType.unknown).length,
        avgDbfs: result.avgDbfs,
        peakDbfs: result.peakDbfs,
        analysisNote: debugNote,
        graphPoints: result.graphPoints,
        segments: segs,
      );
      await SleepSessionStore.append(session);
      if (!mounted) return;
      setState(() => _isAnalyzing = false);
      widget.onSessionSaved(session);
    } catch (e) {
      if (mounted) {
        setState(() {
          _isAnalyzing = false;
          _error = 'Stop/analyze error: $e';
        });
      }
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    _amplitudeSub?.cancel();
    _recorder.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final status = _error ??
        (_isAnalyzing
            ? widget.t.analyzing
            : _isRecording
                ? widget.t.recordingNow
                : widget.t.ready);
    return AppPage(
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          VersionBadge(t: widget.t),
          const SizedBox(height: 14),
          PageTitle(title: widget.t.recordingTitle, sub: widget.t.recordingSub),
          const SizedBox(height: 22),
          Center(
            child: Column(
              children: [
                RecordingPulse(isRecording: _isRecording),
                const SizedBox(height: 18),
                Text(formatDuration(_seconds), style: const TextStyle(color: AppColors.white, fontSize: 46, fontWeight: FontWeight.w900)),
                const SizedBox(height: 10),
                Text(status, textAlign: TextAlign.center, style: TextStyle(color: _error == null ? AppColors.muted : AppColors.red)),
              ],
            ),
          ),
          const SizedBox(height: 20),
          AppCard(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(widget.t.liveSoundMeter, style: const TextStyle(color: AppColors.white, fontWeight: FontWeight.w900)),
                const SizedBox(height: 12),
                LiveMeterBar(db: _liveDb),
                const SizedBox(height: 8),
                Row(
                  children: [
                    Text(_liveDb <= -120 ? 'Live -- dBFS' : 'Live ${_liveDb.toStringAsFixed(1)} dBFS', style: mutedText()),
                    const Spacer(),
                    Text(_peakDb <= -120 ? 'Peak -- dBFS' : 'Peak ${_peakDb.toStringAsFixed(1)} dBFS', style: mutedText()),
                  ],
                ),
              ],
            ),
          ),
          const SizedBox(height: 18),
          Row(
            children: [
              Expanded(
                child: SecondaryButton(
                  text: _isRecording ? widget.t.stop : widget.t.restart,
                  icon: _isRecording ? Icons.stop_rounded : Icons.refresh_rounded,
                  onPressed: _isAnalyzing ? null : (_isRecording ? _stopRecording : _startRecording),
                ),
              ),
              const SizedBox(width: 12),
              Expanded(child: PrimaryButton(text: widget.t.viewReport, icon: Icons.bar_chart_rounded, onPressed: widget.onViewReport)),
            ],
          ),
        ],
      ),
    );
  }
}

class _ReportScreenState extends State<ReportScreen> {
  final AudioPlayer _player = AudioPlayer();
  Timer? _rangeStopTimer;
  bool _isPlaying = false;
  String _playingText = '';

  @override
  void dispose() {
    _rangeStopTimer?.cancel();
    _player.dispose();
    super.dispose();
  }

  Future<void> _stopPlayback() async {
    _rangeStopTimer?.cancel();
    _rangeStopTimer = null;
    await _player.stop();
    if (mounted) {
      setState(() {
        _isPlaying = false;
        _playingText = '';
      });
    }
  }

  Future<void> _playRange({
    required String audioPath,
    required int startSeconds,
    required int durationSeconds,
    required String label,
  }) async {
    final file = File(audioPath);
    if (!file.existsSync()) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(widget.t.audioFileMissingSnack)),
      );
      return;
    }

    await _stopPlayback();

    final safeStart = math.max(0, startSeconds);
    final safeDuration = math.max(1, durationSeconds);

    try {
      await _player.play(
        DeviceFileSource(audioPath),
        position: Duration(seconds: safeStart),
      );

      if (!mounted) return;
      setState(() {
        _isPlaying = true;
        _playingText = label;
      });

      _rangeStopTimer = Timer(Duration(seconds: safeDuration), () async {
        await _stopPlayback();
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _isPlaying = false;
        _playingText = '';
      });
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('${widget.t.cannotPlayAudio}: $e')),
      );
    }
  }

  Future<void> _playFullSession(SleepRecordingSession s) async {
    await _playRange(
      audioPath: s.audioPath,
      startSeconds: 0,
      durationSeconds: math.max(1, s.durationSeconds),
      label: '${widget.t.fullRecording} 00:00 - ${s.durationText}',
    );
  }

  Future<void> _playSegment(SleepRecordingSession s, SleepEventSegment e) async {
    // Add small guard before/after event for easier listening.
    final start = math.max(0, e.startSeconds - 1);
    final end = math.min(s.durationSeconds, e.endSeconds + 1);
    final duration = math.max(1, end - start);
    await _playRange(
      audioPath: s.audioPath,
      startSeconds: start,
      durationSeconds: duration,
      label: '${widget.t.typeLabel(e.type)} ${formatDuration(e.startSeconds)} - ${formatDuration(e.endSeconds)}',
    );
  }

  Widget build(BuildContext context) {
    final s = widget.session;
    if (s == null) {
      return AppPage(
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            VersionBadge(t: widget.t),
            const SizedBox(height: 14),
            PageTitle(title: widget.t.morningReport, sub: widget.t.noSessionSub),
            const SizedBox(height: 16),
            AppCard(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(widget.t.previewTimeline, style: const TextStyle(color: AppColors.white, fontWeight: FontWeight.w900)),
                  const SizedBox(height: 10),
                  LegendRow(t: widget.t),
                  const SizedBox(height: 10),
                  const SizedBox(height: 150, child: SoundTimelineGraph(points: [], segments: [], durationSeconds: 60, forceDemo: true)),
                  const SizedBox(height: 10),
                  Text(widget.t.previewTimelineSub, style: const TextStyle(color: AppColors.muted, fontSize: 12)),
                ],
              ),
            ),
            const SizedBox(height: 18),
            PrimaryButton(text: widget.t.startSleepCheck, icon: Icons.mic_rounded, onPressed: widget.onRecordAgain),
          ],
        ),
      );
    }

    final audioReady = s.hasAudioFile;

class SleepingHeroPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    // room glow
    final bgPaint = Paint()
      ..shader = RadialGradient(
        center: const Alignment(0.8, -0.5),
        radius: 1.1,
        colors: [AppColors.blue.withOpacity(0.36), Colors.transparent],
      ).createShader(Offset.zero & size);
    canvas.drawRect(Offset.zero & size, bgPaint);

    // window
    final windowPaint = Paint()..color = const Color(0xFF0E2C55).withOpacity(0.72);
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(size.width * 0.68, 24, size.width * 0.24, 135), const Radius.circular(20)),
      windowPaint,
    );
    final moonPaint = Paint()..color = AppColors.white.withOpacity(0.86);
    canvas.drawCircle(Offset(size.width * 0.86, 55), 24, moonPaint);
    canvas.drawCircle(Offset(size.width * 0.875, 49), 22, Paint()..color = const Color(0xFF0E2C55));

    // stars
    final starPaint = Paint()..color = AppColors.white.withOpacity(0.48);
    for (final p in [
      Offset(size.width * 0.17, 34),
      Offset(size.width * 0.34, 26),
      Offset(size.width * 0.56, 62),
      Offset(size.width * 0.76, 95),
      Offset(size.width * 0.92, 126),
    ]) {
      canvas.drawCircle(p, 1.5, starPaint);
    }

    final bedY = size.height * 0.68;
    // bed base
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(18, bedY, size.width - 36, 55), const Radius.circular(28)),
      Paint()..color = AppColors.white.withOpacity(0.10),
    );

    // pillow
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(34, bedY - 38, 115, 45), const Radius.circular(22)),
      Paint()..color = AppColors.white.withOpacity(0.88),
    );

    // head / face
    final faceCenter = Offset(118, bedY - 18);
    canvas.drawCircle(faceCenter, 22, Paint()..color = const Color(0xFFE8B99E));

    // hair
    final hairPaint = Paint()
      ..color = const Color(0xFF16121B)
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeWidth = 11;
    canvas.drawArc(Rect.fromCircle(center: Offset(110, bedY - 22), radius: 25), math.pi * 0.68, math.pi * 1.35, false, hairPaint);

    // closed eye
    final eyePaint = Paint()
      ..color = const Color(0xFF4A2E2A)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;
    canvas.drawArc(Rect.fromLTWH(118, bedY - 20, 12, 7), 0, math.pi, false, eyePaint);

    // blanket
    final blanket = Paint()
      ..shader = const LinearGradient(
        colors: [Color(0xFF2D6DFF), Color(0xFF7D5CFF)],
      ).createShader(Rect.fromLTWH(0, bedY - 6, size.width, 80));
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(70, bedY - 4, size.width * 0.72, 70), const Radius.circular(34)),

    );

    // phone glow
    final phoneX = size.width - 92;
    final phoneY = bedY + 8;
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(phoneX - 8, phoneY - 8, 58, 32), const Radius.circular(18)),
      Paint()..color = AppColors.blue.withOpacity(0.16),
    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(phoneX, phoneY, 43, 22), const Radius.circular(7)),
      Paint()..color = const Color(0xFF0B1529),
    );
    canvas.drawRRect(
      RRect.fromRectAndRadius(Rect.fromLTWH(phoneX + 4, phoneY + 4, 35, 14), const Radius.circular(5)),
      Paint()..color = AppColors.blue.withOpacity(0.30),
    );
  }

class TimelinePainter extends CustomPainter {
  final List<SleepGraphPoint> points;
  final List<SleepEventSegment> segments;
  final int durationSeconds;
  final bool forceDemo;

  TimelinePainter({required this.points, required this.segments, required this.durationSeconds, required this.forceDemo});

  @override
  void paint(Canvas canvas, Size size) {
    final bg = Paint()..color = AppColors.navy2.withOpacity(0.90);
    canvas.drawRRect(RRect.fromRectAndRadius(Offset.zero & size, const Radius.circular(18)), bg);

    final left = 32.0;
    final right = size.width - 12;
    final top = 14.0;
    final bottom = size.height - 24;
    final w = math.max(1.0, right - left);
    final h = math.max(1.0, bottom - top);

    // grid and labels
    final gridPaint = Paint()
      ..color = AppColors.white.withOpacity(0.08)
      ..strokeWidth = 1;
    final labelStyle = TextStyle(color: AppColors.white.withOpacity(0.38), fontSize: 9);
    for (final db in [-80, -60, -40, -20, 0]) {
      final y = _dbToY(db.toDouble(), top, h);
      canvas.drawLine(Offset(left, y), Offset(right, y), gridPaint);
      _drawText(canvas, '${db}dB', Offset(4, y - 6), labelStyle);
    }

    final data = (forceDemo || points.isEmpty) ? _demo(durationSeconds <= 0 ? 60 : durationSeconds) : points;
    final dur = math.max(1, durationSeconds <= 0 ? data.last.offsetSeconds : durationSeconds);
    final bw = math.max(2.0, w / math.max(1, data.length) * 0.70);

    for (final p in data) {
      final x = left + (p.offsetSeconds / dur) * w;
      final y = _dbToY(p.peakDbfs, top, h);
      final c = colorForType(p.type).withOpacity(p.type == SoundType.background ? 0.42 : 0.96);
      final topY = math.min(bottom - 2.0, y);
      canvas.drawRRect(
        RRect.fromRectAndRadius(Rect.fromLTWH(x, topY, bw, bottom - topY), const Radius.circular(3)),
        Paint()..color = c,
      );
    }

    final segs = forceDemo ? _demoSegments(dur) : segments;
    for (final s in segs) {
      final x1 = left + (s.startSeconds / dur) * w;
      final x2 = left + (s.endSeconds / dur) * w;
      final c = colorForType(s.type);
      canvas.drawRRect(
        RRect.fromRectAndRadius(Rect.fromLTRB(x1, top + 2, math.max(x1 + 5, x2), top + 10), const Radius.circular(5)),
        Paint()..color = c,
      );
    }

    _drawText(canvas, '0:00', Offset(left, bottom + 5), labelStyle);
    _drawText(canvas, formatDuration(dur), Offset(right - 40, bottom + 5), labelStyle);
  }

  double _dbToY(double db, double top, double h) {
    final c = db.clamp(-80.0, 0.0).toDouble();
    final n = (c + 80) / 80.0;
    return top + h * (1 - n);
  }

  void _drawText(Canvas canvas, String text, Offset offset, TextStyle style) {
    final tp = TextPainter(
      text: TextSpan(text: text, style: style),
      textDirection: TextDirection.ltr,
    )..layout();
    tp.paint(canvas, offset);
  }

  List<SleepGraphPoint> _demo(int d) {
    final out = <SleepGraphPoint>[];
    for (var i = 0; i <= d; i += 2) {
      var db = -76.0 + math.sin(i * 0.45) * 2;
      var type = SoundType.background;
      if (i >= d * 0.14 && i <= d * 0.24) {
        db = -34 + math.sin(i).abs() * 7;
        type = SoundType.speech;
      } else if (i >= d * 0.36 && i <= d * 0.53) {
        db = -42 + math.sin(i * 0.75).abs() * 8;
        type = SoundType.possibleSnore;
      } else if (i >= d * 0.62 && i <= d * 0.67) {
        db = -12;
        type = SoundType.impact;
      } else if (i >= d * 0.78 && i <= d * 0.90) {
        db = -47 + math.sin(i * 0.80).abs() * 7;
        type = SoundType.unknown;
      }
      out.add(SleepGraphPoint(offsetSeconds: i, avgDbfs: db - 3, peakDbfs: db, type: type));
    }
    return out;
  }

  List<SleepEventSegment> _demoSegments(int d) {
    return [
      SleepEventSegment(id: 1, startSeconds: (d * 0.14).floor(), endSeconds: (d * 0.24).floor(), type: SoundType.speech, avgDbfs: -34, peakDbfs: -20, crestDb: 8, stdDb: 7, detail: 'demo'),
      SleepEventSegment(id: 2, startSeconds: (d * 0.36).floor(), endSeconds: (d * 0.53).floor(), type: SoundType.possibleSnore, avgDbfs: -42, peakDbfs: -28, crestDb: 10, stdDb: 5, detail: 'demo'),
      SleepEventSegment(id: 3, startSeconds: (d * 0.62).floor(), endSeconds: (d * 0.67).floor(), type: SoundType.impact, avgDbfs: -22, peakDbfs: -9, crestDb: 16, stdDb: 11, detail: 'demo'),
      SleepEventSegment(id: 4, startSeconds: (d * 0.78).floor(), endSeconds: (d * 0.90).floor(), type: SoundType.unknown, avgDbfs: -47, peakDbfs: -30, crestDb: 12, stdDb: 9, detail: 'demo'),
    ];
  }

  @override
  bool shouldRepaint(covariant TimelinePainter oldDelegate) => true;
}

TextStyle mutedText([double size = 12.5]) {
  return TextStyle(color: AppColors.white.withOpacity(0.58), fontSize: size, height: 1.35);
}

String formatDuration(int seconds) {
  final h = seconds ~/ 3600;
  final m = (seconds % 3600) ~/ 60;
  final s = seconds % 60;
  if (h > 0) {
    return '${h.toString().padLeft(2, '0')}:${m.toString().padLeft(2, '0')}:${s.toString().padLeft(2, '0')}';
  }
  return '${m.toString().padLeft(2, '0')}:${s.toString().padLeft(2, '0')}';
}
